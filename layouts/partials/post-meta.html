{{/* post-meta.html - the metadata strip. Takes a dict of {page, context}, where context is "list" or "single". Which items appear is controlled by params.features and params.post. */}} {{- $p := .page -}} {{- $ctx := .context | default "list" -}} {{- $feat := $p.Site.Params.features | default dict -}} {{- $postCfg := $p.Site.Params.post | default dict -}} {{- $dateFmt := $p.Site.Params.dateFormat | default "Jan 2, 2006" -}} {{- $author := $p.Params.author | default $p.Site.Params.author -}}
{{- with $author }} {{ with partial "svg-icon.html" "pencil" }}{{ . | safeHTML }}{{ end }} Author: {{ . }} {{- end }} {{- if $feat.publishDate | default true }} {{ with partial "svg-icon.html" "calendar" }}{{ . | safeHTML }}{{ end }} Published: {{ $p.Date.Format $dateFmt }} {{- end }} {{- if $p.Lastmod.After $p.Date }} {{- if $postCfg.showLastmod | default true }} {{ with partial "svg-icon.html" "refresh" }}{{ . | safeHTML }}{{ end }} Last updated: {{ $p.Lastmod.Format $dateFmt }} {{- end }} {{- end }} {{- if and ($postCfg.readingTime | default true) $p.IsPage }} {{ with partial "svg-icon.html" "clock" }}{{ . | safeHTML }}{{ end }} Reading time: {{ $p.ReadingTime }} min read {{- end }} {{- if and ($postCfg.wordCount | default false) $p.IsPage }} {{ with partial "svg-icon.html" "list" }}{{ . | safeHTML }}{{ end }} Words: {{ $p.WordCount }} words {{- end }} {{- if eq $ctx "single" }} {{- with $p.Params.categories }} {{ with partial "svg-icon.html" "folder" }}{{ . | safeHTML }}{{ end }} Categories: {{- range $i, $c := . }}{{ if $i }}, {{ end -}} {{ $c }} {{- end }} {{- end }} {{- end }}
