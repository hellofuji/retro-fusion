{{/* share.html - share buttons. Plain links and mailto:, no third-party script. Toggle with params.post.share, or `share: false` on a page. The full URLs are built up front and passed through safeURL: Go's html/template re-escapes a value inside an href, which would otherwise turn an already-encoded %3A into %253A. */}} {{- $postCfg := .Site.Params.post | default dict -}} {{- if and ($postCfg.share | default true) (ne .Params.share false) }} {{- $url := .Permalink -}} {{- $title := .Title -}} {{- $eq := urlquery $url -}} {{- $et := urlquery $title -}} {{- $urls := dict "x" (printf "https://twitter.com/intent/tweet?url=%s&text=%s" $eq $et) "facebook" (printf "https://www.facebook.com/sharer/sharer.php?u=%s" $eq) "linkedin" (printf "https://www.linkedin.com/sharing/share-offsite/?url=%s" $eq) "reddit" (printf "https://www.reddit.com/submit?url=%s&title=%s" $eq $et) "telegram" (printf "https://t.me/share/url?url=%s&text=%s" $eq $et) "email" (printf "mailto:?subject=%s&body=%s" $et $eq) -}} {{- $nets := $postCfg.shareNetworks | default (slice "x" "facebook" "linkedin" "reddit" "mastodon" "email") -}}
Share: {{- range $nets }} {{- $key := . -}} {{- $icon := partial "svg-icon.html" $key -}} {{- with index $urls $key }} {{- with $icon }} {{ . | safeHTML }}{{ end }} {{ $key | humanize | title }} {{- else }} {{- if eq $key "mastodon" }} {{- with $icon }} {{ . | safeHTML }}{{ end }} Mastodon {{- else if eq $key "copy" }} {{- else if eq $key "print" }} {{- end }} {{- end }} {{- end }}
{{- end }}
